# Go to your Desktop
cd $HOME\Desktop

# Create an Electron Forge project that already uses Vite
npx create-electron-app@latest cardforge --template=vite

# Enter the new project folder
cd cardforge

# Install React, ReactDOM, and the Vite React plugin
npm install react react-dom
npm install --save-dev @vitejs/plugin-react@4

# Create a small React app folder inside the renderer source folder
New-Item -ItemType Directory -Force -Path "src\renderer\src" | Out-Null

# Start the desktop app
npm start
